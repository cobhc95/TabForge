cmake_minimum_required(VERSION 3.20)
cmake_policy(SET CMP0091 NEW) # CMAKE_MSVC_RUNTIME_LIBRARY
project(tfportable LANGUAGES C CXX)

# tfportable: miniaudio (audio) and RtMidi (MIDI) behind one small C ABI (tfportable.h). Built for the Linux and macOS
# (Apple Silicon) builds, and on Windows only as an opt-in dev library (the shipped Windows build keeps WASAPI/ASIO and
# winmm and does not contain it). Both libraries are PINNED to an exact commit and fetched; nothing is patched.
set(CMAKE_MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_C_STANDARD 11)
set(CMAKE_POSITION_INDEPENDENT_CODE ON)

set(TFPORTABLE_MINIAUDIO_TAG    "0.11.25")
set(TFPORTABLE_MINIAUDIO_COMMIT "9634bedb5b5a2ca38c1ee7108a9358a4e233f14d")
set(TFPORTABLE_RTMIDI_TAG       "6.0.0")
set(TFPORTABLE_RTMIDI_COMMIT    "1e5b49925aa60065db52de44c366d446a902547b")
set(MINIAUDIO_DIR "" CACHE PATH "Optional: an existing miniaudio checkout (the folder with miniaudio.h)")
set(RTMIDI_DIR "" CACHE PATH "Optional: an existing RtMidi checkout (the folder with RtMidi.cpp)")

include(FetchContent)
if(NOT MINIAUDIO_DIR)
    FetchContent_Declare(miniaudio GIT_REPOSITORY https://github.com/mackron/miniaudio.git GIT_TAG ${TFPORTABLE_MINIAUDIO_COMMIT} SOURCE_SUBDIR _fetch_only)
    FetchContent_MakeAvailable(miniaudio)
    set(MINIAUDIO_DIR "${miniaudio_SOURCE_DIR}")
endif()
if(NOT RTMIDI_DIR)
    FetchContent_Declare(rtmidi GIT_REPOSITORY https://github.com/thestk/rtmidi.git GIT_TAG ${TFPORTABLE_RTMIDI_COMMIT} SOURCE_SUBDIR _fetch_only)
    FetchContent_MakeAvailable(rtmidi)
    set(RTMIDI_DIR "${rtmidi_SOURCE_DIR}")
endif()
if(NOT EXISTS "${MINIAUDIO_DIR}/miniaudio.h")
    message(FATAL_ERROR "miniaudio.h not found in ${MINIAUDIO_DIR}")
endif()
if(NOT EXISTS "${RTMIDI_DIR}/RtMidi.cpp")
    message(FATAL_ERROR "RtMidi.cpp not found in ${RTMIDI_DIR}")
endif()

add_library(tfportable SHARED tfaudio.c tfmidi.cpp tfportable.h "${RTMIDI_DIR}/RtMidi.cpp")
target_include_directories(tfportable PRIVATE "${MINIAUDIO_DIR}" "${RTMIDI_DIR}" "${CMAKE_CURRENT_SOURCE_DIR}")
set_target_properties(tfportable PROPERTIES C_VISIBILITY_PRESET hidden CXX_VISIBILITY_PRESET hidden)

if(WIN32)
    target_compile_definitions(tfportable PRIVATE __WINDOWS_MM__ NOMINMAX WIN32_LEAN_AND_MEAN _CRT_SECURE_NO_WARNINGS)
    target_link_libraries(tfportable PRIVATE winmm)
elseif(APPLE)
    target_compile_definitions(tfportable PRIVATE __MACOSX_CORE__)
    target_link_libraries(tfportable PRIVATE "-framework CoreMIDI" "-framework CoreAudio" "-framework CoreFoundation" "-framework AudioToolbox")
    set(CMAKE_OSX_ARCHITECTURES "arm64" CACHE STRING "Apple Silicon only")
else()
    find_package(ALSA)
    if(ALSA_FOUND)
        target_compile_definitions(tfportable PRIVATE __LINUX_ALSA__)
        target_include_directories(tfportable PRIVATE ${ALSA_INCLUDE_DIRS})
        target_link_libraries(tfportable PRIVATE ${ALSA_LIBRARIES})
    else()
        message(WARNING "ALSA headers not found (install libasound2-dev): MIDI is built as the dummy API")
        target_compile_definitions(tfportable PRIVATE __RTMIDI_DUMMY__)
    endif()
    target_link_libraries(tfportable PRIVATE pthread dl m)
    target_link_options(tfportable PRIVATE -static-libstdc++ -static-libgcc)   # no dependence on the system libstdc++/libgcc version
endif()

# miniaudio loads PulseAudio, ALSA and JACK at run time (dlopen), so no sound library is a hard dependency on Linux.
install(TARGETS tfportable RUNTIME DESTINATION . LIBRARY DESTINATION .)
